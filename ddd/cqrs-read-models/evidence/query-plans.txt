## group-by
SELECT s.id, SUM(r.status='CONFIRMED'), SUM(r.status='WAITLISTED'), SUM(r.paid_cents) FROM session s JOIN registration r ON r.session_id = s.id WHERE s.event_id = 1 GROUP BY s.id ORDER BY s.id
